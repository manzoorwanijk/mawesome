import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';
import ToolCard from '../../components/ToolCard.astro';
import { actions, tools } from '../../tools.ts';
import { toolVersion } from '../../lib/tool-versions.ts';

<h2 id="packages">📦 npm packages</h2>

<div class="tool-grid">
	{tools.map((tool) => (
		<ToolCard
			name={tool.npm}
			href={`/${tool.slug}/`}
			icon={tool.icon}
			version={toolVersion(tool.npm)}
			versionHref={`https://www.npmjs.com/package/${tool.npm}`}
			tagline={tool.tagline}
			code={tool.quickstart}
			lang="sh"
			links={[
				{ label: 'Documentation', href: `/${tool.slug}/` },
				...(tool.playground ? [{ label: '⚡ Playground', href: `/${tool.slug}/playground/` }] : []),
				{ label: 'npm', href: `https://www.npmjs.com/package/${tool.npm}` },
				{ label: 'Source', href: `https://github.com/mawesomedev/mawesome/tree/main/packages/${tool.slug}` },
			]}
		/>
	))}
</div>

<h2 id="actions">⚙️ GitHub Actions</h2>

Each action is released to its own mirror repository, which is what a workflow references.

<div class="tool-grid">
	{actions.map((action) => {
		const version = toolVersion(action.workspace);
		return (
			<ToolCard
				name={action.repo}
				href={action.docs}
				icon={action.icon}
				version={version}
				versionHref={`https://github.com/${action.repo}/tree/v${version}`}
				tagline={action.tagline}
				code={`- uses: ${action.repo}@<sha> # v${version}\n  with:\n    base: main`}
				lang="yaml"
				links={[
					{ label: 'Documentation', href: action.docs },
					{ label: 'Mirror', href: `https://github.com/${action.repo}` },
				]}
			/>
		);
	})}
</div>
